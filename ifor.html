4-3. __iter__로 클래스를 반복 가능하게

class MyCounter:
    def __init__(self, low, high):
        self.current = low
        self.high = high
    def __iter__(self):
        return self
    def __next__(self):
        if self.current > self.high:
            raise StopIteration
        else:
            self.current += 1
            return self.current - 1

counter = MyCounter(1, 3)
for num in counter:
    print(num)  # 1, 2, 3 출력

class User:

        self.id = user_id
        self.name = name

    def create_users_from_lists(cls, ids, names):
        return [cls(uid, name) for uid, name in zip(ids, names)]

class Position:

        self.x = x
        self.y = y
